* System Bootstrap (Auto-run)
This script handles Arch packages and repositories. It is strictly idempotent and executes automatically via our custom tangling engine.

#+begin_src bash :tangle ~/.local/bin/desktop-sync :shebang "#!/usr/bin/env bash"
# Enable strict error handling
set -euo pipefail

echo "--- Starting System Sync ---"

# 1. Update/Install Packages (Idempotent)
PACKAGES=(atuin bat imv hyprland waybar)

# pacman -T returns the list of missing packages. If none are missing, it returns empty.
MISSING_PKGS=$(pacman -T "${PACKAGES[@]}" || true)

if [ -n "$MISSING_PKGS" ]; then
    echo "Installing missing packages: $MISSING_PKGS"
    sudo pacman -S --needed --noconfirm $MISSING_PKGS
else
    echo "All core packages are already installed."
fi

# 2. Clone or Update Repositories (Idempotent)
sync_repo() {
    local url=$1
    local dest=$2
    if [ ! -d "$dest" ]; then
        echo "Cloning $url..."
        git clone "$url" "$dest"
    else
        echo "Updating $dest..."
        # Perform a fast-forward pull; if it fails (e.g., local changes), echo a warning but don't crash
        git -C "$dest" pull --ff-only || echo "Warning: Could not cleanly update $dest."
    fi
}

sync_repo "https://github.com/tmux-plugins/tpm" "$HOME/.tmux/plugins/tpm"

echo "--- Sync Finished Successfully ---"
#+end_src

* The Tangling Engine (Core Logic)
This block is evaluated automatically when you open the file. It defines a robust function that expands all includes into a temporary buffer and tangles the unified tree.

#+name: define-tangler
#+begin_src emacs-lisp :tangle no :results silent
(defun my/tangle-all-and-sync ()
  "Expands #+INCLUDEs natively, tangles, and runs the sync script."
  (interactive)
  (let ((master-file (buffer-file-name)))
    (message "Tangling Master Org and all included modules...")
    (with-temp-buffer
      (insert-file-contents master-file)
      (org-mode)
      ;; Trick Emacs into thinking this temp buffer is the real file to resolve relative paths
      (setq buffer-file-name master-file)
      ;; Natively expand all #+INCLUDE directives
      (org-export-expand-include-keyword)
      ;; Tangle the unified buffer
      (org-babel-tangle))
    (message "Tangling complete. Starting sync script asynchronously...")
    ;; Run the script and pipe output to a dedicated Emacs buffer
    (start-process-shell-command "desktop-sync" "*Desktop Sync Output*" "~/.local/bin/desktop-sync")))
#+end_src
