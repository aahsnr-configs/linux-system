// Niri-Native Keybindings
// Only actions that call niri compositor actions live here.
// Application spawns → ~/.config/swhkd/swhkdrc
// Keychord submaps   → ~/.config/xremap/config.yml
// Mouse/complex      → ~/.config/kanata/kanata.kbd
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

binds {

    // ─── COMPOSITOR CONTROL ─────────────────────────────────────────────────
    Super+Shift+E { quit; }                          // Exit niri (with dialog)
    Super+Shift+R { spawn-sh "niri msg action reload-config"; }   // Live-reload
    // Lock screen is handled by Noctalia Shell via swhkd (Super+Alt+L)

    // ─── WINDOW MANAGEMENT ──────────────────────────────────────────────────
    Super+Q         { close-window; }
    Super+Space     { toggle-window-floating; }
    Super+Shift+Space { fullscreen-window; }
    // True edge-to-edge maximise (niri 25.11+)
    Super+Shift+M   { maximize-window-to-edges; }
    // Center the focused floating window / column
    Super+C         { center-column; }

    // ─── FOCUS (Arrow keys + hjkl) ──────────────────────────────────────────
    Super+Left      { focus-column-left; }
    Super+Right     { focus-column-right; }
    Super+Up        { focus-window-up; }
    Super+Down      { focus-window-down; }

    Super+H         { focus-column-left; }
    Super+L         { focus-column-right; }
    Super+K         { focus-window-up; }
    Super+J         { focus-window-down; }

    // Focus first/last column on the workspace
    Super+Home      { focus-column-first; }
    Super+End       { focus-column-last; }

    // ─── MOVE WINDOW / COLUMN ───────────────────────────────────────────────
    Super+Shift+Left   { move-column-left; }
    Super+Shift+Right  { move-column-right; }
    Super+Shift+Up     { move-window-up; }
    Super+Shift+Down   { move-window-down; }

    Super+Shift+H      { move-column-left; }
    Super+Shift+L      { move-column-right; }
    Super+Shift+K      { move-window-up; }
    Super+Shift+J      { move-window-down; }

    Super+Shift+Home   { move-column-to-first; }
    Super+Shift+End    { move-column-to-last; }

    // ─── RESIZE ─────────────────────────────────────────────────────────────
    Super+Ctrl+Left    { set-column-width "-5%"; }
    Super+Ctrl+Right   { set-column-width "+5%"; }
    Super+Ctrl+Up      { set-window-height "-5%"; }
    Super+Ctrl+Down    { set-window-height "+5%"; }

    // Cycle preset widths (Mod+R) / heights (Mod+Shift+W)
    Super+R            { switch-preset-column-width; }
    Super+Shift+W      { switch-preset-window-height; }
    // Expand to fill remaining width
    Super+F            { maximize-column; }

    // ─── COLUMN STACKING ────────────────────────────────────────────────────
    Super+BracketLeft  { consume-or-expel-window-left; }
    Super+BracketRight { consume-or-expel-window-right; }
    Super+Comma        { consume-window-into-column; }
    Super+Period       { expel-window-from-column; }

    // ─── TABBED COLUMNS (niri 25.11+) ────────────────────────────────────────
    Super+T            { toggle-column-tabbed-display; }

    // ─── OVERVIEW (Bird's eye) ───────────────────────────────────────────────
    Super+Grave        { toggle-overview; }

    // ─── ALT-TAB ─────────────────────────────────────────────────────────────
    Alt+Tab repeat=false { switch-focus-between-floating-and-tiled; }

    // ─── SCROLL ACROSS COLUMNS (mouse wheel / touchpad) ──────────────────────
    Super+WheelScrollDown cooldown-ms=150 { focus-workspace-down; }
    Super+WheelScrollUp   cooldown-ms=150 { focus-workspace-up; }
    Super+WheelScrollRight               { focus-column-right; }
    Super+WheelScrollLeft                { focus-column-left; }

    Ctrl+Super+WheelScrollDown cooldown-ms=150 { move-column-to-workspace-down; }
    Ctrl+Super+WheelScrollUp   cooldown-ms=150 { move-column-to-workspace-up; }

    // ─── FOCUS WORKSPACE ─────────────────────────────────────────────────────
    Super+1 { focus-workspace "browser"; }
    Super+2 { focus-workspace "dev"; }
    Super+3 { focus-workspace "term"; }
    Super+4 { focus-workspace "research"; }
    Super+5 { focus-workspace "tex"; }
    Super+6 { focus-workspace "comms"; }
    Super+7 { focus-workspace "media"; }
    Super+8 { focus-workspace "scratch"; }
    Super+9 { focus-workspace "stage"; }

    // Relative workspace navigation (raw fallback; smart cycling is in swhkdrc)
    Ctrl+Alt+Down { focus-workspace-down; }
    Ctrl+Alt+Up   { focus-workspace-up; }

    // ─── MOVE COLUMN TO WORKSPACE ────────────────────────────────────────────
    Super+Shift+1 { move-column-to-workspace "browser"; }
    Super+Shift+2 { move-column-to-workspace "dev"; }
    Super+Shift+3 { move-column-to-workspace "term"; }
    Super+Shift+4 { move-column-to-workspace "research"; }
    Super+Shift+5 { move-column-to-workspace "tex"; }
    Super+Shift+6 { move-column-to-workspace "comms"; }

;; Kanata Configuration
;;
;; Purpose: Handle input remapping that neither niri nor swhkd can do natively.
;; Specifically:
;;   1. Super key tap → open launcher (when tapped alone, not as a modifier).
;;      niri and swhkd have no tap-hold concept for modifier keys.
;;   2. Caps Lock dual-role: held = Ctrl, tapped = Escape.
;;      Complements the XKB ctrl:nocaps option; this goes further by making
;;      a solo-tap of Caps emit Escape (extremely useful in Vim / Neovim).
;;
;; IMPORTANT: Mouse drag-to-move (Super+LMB) and drag-to-resize (Super+RMB)
;; are handled NATIVELY by niri for floating windows — no kanata needed.
;; Wheel-scroll-workspace bindings live in conf.d/binds.kdl.
;; ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

(defcfg
  ;; Leave this disabled since we are using the secure key-chord method
  ;; to trigger the launcher via swhkdrc.
  ;; danger-enable-cmd yes
)

;; ── Source keys to intercept ────────────────────────────────────────────────
;; Only intercept Caps Lock and Left Meta (Super).
;; Everything else passes through untouched.
(defsrc
  caps lmet
)

;; ── Aliases ──────────────────────────────────────────────────────────────────
(defalias
  ;; Caps Lock:
  ;;   held  → Left Ctrl
  ;;   tapped alone → Escape
  caps-dual (tap-hold-release 200 200 esc lctl)

  ;; Left Meta (Super):
  ;;   held  → Left Meta (normal modifier behaviour)
  ;;   tapped alone → Emits M-d (Left Meta + D)
  ;;   Your swhkdrc catches Super+D and runs: qs -c noctalia-shell ipc call launcher toggle
  super-dual (tap-hold-release 200 200 M-d lmet)
)

;; ── Default Layer ────────────────────────────────────────────────────────────
;; Map the source keys to our dual-function aliases.
(deflayer default
  @caps-dual @super-dual
)
